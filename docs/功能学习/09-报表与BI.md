```python
def crud(prefix, viewset, extras=None):
    def _rp(s): return s if prefix else s.lstrip('/')
    def _join(s): return f'{prefix}/{s}' if prefix else s
    list_view = viewset.as_view({'get': 'list', 'post': 'create', 'delete': 'delete'})  # 三个动作共用一条路径
    item_view = viewset.as_view({'put': 'update'})                                     # 更新单独一条路径
    patterns = [
        path(_join('list'), list_view),                # /datasource/list
        path(_join('create'), list_view),              # /datasource/create
        path(_join('delete'), list_view),              # /datasource/delete
        path(_join('update/<int:pk>'), item_view),     # /datasource/update/{id}
    ]
    for suffix, methods in (extras or {}).items():     # 额外业务动作，如 status
        patterns.append(path(_join(suffix), viewset.as_view(methods)))
    return patterns
```

```python
@property
def permission_classes(self):
    prefix = self.perm_prefix            # 子类配置的权限前缀，此处为 rp
    self.perm_map = {
        'list': f'{prefix}:list',        # 列表 → rp:list
        'create': f'{prefix}:create',    # 新增 → rp:create
        'update': f'{prefix}:update',    # 编辑 → rp:update

    return [PermissionCheck]             # 只挂自定义权限类，登录校验在其内部完成

def get_queryset(self):
    # 统一过滤逻辑删除：is_deleted=0，相当于 MyBatis-Plus 的 @TableLogic
    return self.model.objects.filter(is_deleted=0).order_by(self.order_by)
```

```python
# backend/common/views.py:50-61
def apply_filters(self, queryset, request):
    for param, lookup in self.filter_map.items():                 # 遍历子类声明的筛选项
        value = request.query_params.get(param)                   # 从 URL 查询串取参数
        if value not in (None, ''):                               # 空值/null 视为不筛选
            queryset = queryset.filter(**{f'{param}__{lookup}': value})  # 动态拼 ds_name__icontains 等
    return queryset

def list(self, request):
    queryset = self.apply_filters(self.get_queryset(), request)   # 先过滤再分页，保证 total 准确
    return StandardPagination(request).paginate(queryset, self.get_serializer_class())  # 统一分页响应
```

权限校验真实实现（登录 + 按钮权限）：

```python
# backend/common/permissions.py:43-58
class PermissionCheck(LoginRequired):
    def has_permission(self, request, view):
        if not super().has_permission(request, view):   # 先校验登录，未登录抛 401
            return False
        perm_map = getattr(view, 'perm_map', None)      # 取视图上的 action → 权限标识映射
        if not perm_map:
            return True                                 # 未声明映射视为无需细粒度权限
        required = perm_map.get(view.action)            # 当前 action 对应的权限标识，如 rp:list
        if not required:
            return True                                 # 该 action 未配置权限要求，直接放行
        perms = get_user_perms(request.user_info)       # 从用户-角色-菜单关联里取权限集合
        if '*:*:*' in perms or required in perms:       # 超管通配或已明确授权
            return True
        raise drf_exceptions.PermissionDenied(f'没有操作权限：{required}')  # 403
```

`get_user_perms` 用一条 SQL 关联 `sys_user_role → sys_role → sys_role_menu → sys_menu`，并把 `role_key=admin` 的角色直接加通配权限（`backend/common/permissions.py:18-38`）。

```python
def _resolve_meta(category_id, ds_id):
    """解析分类名与数据源名，供预览页展示来源信息。"""
    category_name = ''
    ds_name = ''
    if category_id:
        # values_list(..., flat=True).first() 只查一列，等价于 select category_name limit 1
        category_name = RpCategory.objects.filter(pk=category_id).values_list('category_name', flat=True).first() or ''
    if ds_id:
        ds_name = RpDatasource.objects.filter(pk=ds_id).values_list('ds_name', flat=True).first() or ''
    return category_name, ds_name

def _build(template_json, name, seed, template_id, key, category_name, ds_name, version):
    """按模板配置构建报表数据包，四种视图（柱/线/饼/表）统一结构。"""
    core = build_chart_data(parse_config(template_json), name, seed, 'bar')  # 核心图表数据
    return {
        'id': template_id, 'name': name, 'key': key,                          # 模板标识
        'category_name': category_name, 'ds_name': ds_name,                   # 来源信息
        'version': version,                                                   # 版本号
        'chart_label': CHART_LABELS.get(core['chart'], ''),                   # 中文图表名
        **core,                                                               # 展开 chart/categories/series/rows 等
        'updated_at': time.strftime('%Y-%m-%d %H:%M:%S'),                     # 本次渲染时间
        'record': None,                                                       # 预览没有生成记录

def build_report(template):
    """根据报表模板生成预览数据。"""
    category_name, ds_name = _resolve_meta(template.category_id, template.ds_id)
    return _build(template.template_json, template.template_name, template.id, template.id,
                  template.template_key or '', category_name, ds_name, template.version)
```

```python
def run_readonly_sql(sql, limit=100):
    """执行只读 SQL：仅允许单条 SELECT，失败或无结果返回 None。"""
    if not isinstance(sql, str):
        return None                                # 非字符串直接拒绝
    text = sql.strip().rstrip(';').strip()         # 去掉首尾空白与结尾分号
    lowered = text.lower()
    if not lowered.startswith('select') or ';' in text:  # 只允许单条 SELECT
        return None
    if any(word in f' {lowered} ' for word in BLOCK_WORDS):  # 含 insert/update/delete/drop 等写关键字直接拒绝
        return None
    try:
        with connection.cursor() as cursor:        # 复用默认数据库连接
            cursor.execute(text)
            columns = [str(col[0]) for col in cursor.description or []]  # 列名
            values = cursor.fetchmany(limit)       # 最多取 limit 行
    except Exception:
        return None                                # SQL 出错静默退回演示数据，保证预览不 500
```

```python
# backend/apps/report/views.py:66-88
@action(detail=True, methods=['post'])  # 必须是 POST，前端用 request.post
def generate(self, request, pk=None):
    """手动生成报表：落生成记录并直接返回本次渲染结果。"""
    tpl = RpTemplate.objects.filter(pk=pk, is_deleted=0).first()  # 只允许对未删除模板生成
    if not tpl:
        return fail('模板不存在', code=404)                        # 业务 404
    user = getattr(request, 'user_info', None)                    # JWT 认证类挂上去的当前用户
    now = time.strftime('%Y-%m-%d %H:%M:%S')                      # 统一时间字符串，与手写 SQL 入库格式一致
    data = request.data or {}                                     # 兼容空请求体
    param = data.get('param_json') or data.get('param') or ''     # 前端可传生成参数，未传则为空
    started = time.time()                                         # 开始计时
    report = build_report(tpl)                                    # 先渲染数据，耗时计入生成记录
    cost = int((time.time() - started) * 1000)                    # 毫秒级耗时
    record = RpGenerateRecord.objects.create(                     # 每次生成都落一条记录
        template_id=tpl.id, template_name=tpl.template_name,      # 模板 id + 名称快照
        param_json=str(param), gen_status=1, gen_type='manual',   # 状态固定成功，类型固定手动
        gen_user_id=getattr(user, 'id', None),                    # 操作者 id，可能为空
        cost_time=max(cost, 1), gen_time=now,                     # 至少 1ms，避免显示 0
        create_by=user.username if user else '',                  # 审计字段：创建人账号
        create_time=now, update_by=user.username if user else '', # 更新人初始与创建人一致
        update_time=now, is_deleted=0,                            # 未删除
    )
    report['record'] = RpGenerateRecordSerializer(record).data    # 把生成记录附在返回 payload 里
    return ok(report, '报表生成成功')
```

关键点：

1. **没有异常兜底**：`build_report` 内部对 JSON/ SQL 都做了容错，但若出现未捕获异常，`gen_status=2`、`error_msg` 的失败分支永远不会写入。
2. **没有事务包裹**：渲染与写记录不是一个原子操作（Django 默认 autocommit）。
3. **耗时只统计渲染**，不含写库时间。

```python
    serializer_class = RpGenerateRecordSerializer

        'gen_status': 'exact',                # 生成状态精确匹配
        'gen_type': 'exact',                  # 触发方式精确匹配
    }
    order_by = '-id'                          # 最新生成在前
    module_name = '报表生成记录'

    @action(detail=True, methods=['get'])
    def preview(self, request, pk=None):
```

回放的核心逻辑：

```python
# backend/apps/report/render.py:42-58
def build_record_report(record):
    """根据生成记录回放报表结果：模板被删除时用记录快照生成演示数据。"""
    template = RpTemplate.objects.filter(pk=record.template_id).first()  # 注意：没有过滤 is_deleted
    if template:
        category_name, ds_name = _resolve_meta(template.category_id, template.ds_id)
        payload = _build(template.template_json, record.template_name or template.template_name,
                         template.id, template.id, template.template_key or '',
                         category_name, ds_name, template.version)       # 用模板当前配置渲染
    else:
        payload = _build('', record.template_name or '已删除的报表模板', record.template_id or record.id,
                         record.template_id, '', '', '', 0)              # 模板物理删除时兜底
    payload['record'] = {
        'id': record.id, 'gen_status': record.gen_status, 'gen_type': record.gen_type,
        'cost_time': record.cost_time, 'gen_time': record.gen_time,
        'error_msg': record.error_msg or '',
        'param': record.param_json or '',
    }                                                                    # 附上本次生成记录信息
    return payload
```

要点：

- 如果模板只是**逻辑删除**（`is_deleted=1`），`filter(pk=...)` 仍能查到，会继续用模板当前配置渲染，和注释「模板被删除时用记录快照」并不完全一致。
- 回放用的是**模板现在的** `template_json`，不是生成当时的配置；模板改过配置后，历史记录回放结果会变化。`rp_generate_record` 没有存模板 JSON 快照。

```python
def build_screen(screen):
    """构建大屏展示数据包：画布尺寸、背景、组件列表及各组件数据。"""
    cfg = parse_config(screen.screen_config)                    # 解析 JSON，非法配置回退空字典

    height = int(cfg.get('height') or 1080)                     # 设计稿高度，默认 1080
    widgets = cfg.get('widgets') if isinstance(cfg.get('widgets'), list) else []
    if not widgets:
        widgets = DEFAULT_WIDGETS                               # 未配置组件时给 4 个演示组件
    dataset_ids = [widget.get('dataset_id') for widget in widgets
                   if isinstance(widget, dict) and widget.get('dataset_id')]  # 收集组件引用的数据集
    datasets = {row.id: row for row in BiDataset.objects.filter(id__in=dataset_ids, is_deleted=0)}
    payload_widgets = []
    for index, widget in enumerate(widgets):
        if not isinstance(widget, dict):
            continue
        wtype = normalize_chart(widget.get('type'), 'bar')
        wtype = wtype if wtype in WIDGET_TYPES else 'bar'

            'type': wtype,

            'subtitle': str(widget.get('subtitle') or ''),
            **_geometry(widget, index, width),                  # 有 x/y/w/h 用配置，否则 4 列自动排版
            'unit': str(widget.get('unit') or ''),
            'data': _widget_data(widget, index, screen.id or 0, datasets),  # 内联 → 数据集 → 演示数据
        })
    return {
        'screen_id': screen.id, 'screen_name': screen.screen_name,
        'status': screen.status, 'refresh_sec': screen.refresh_sec or 0,
        'share_url': screen.share_url or '', 'share_password': screen.share_password or '',
        'width': width, 'height': height, 'background': str(cfg.get('background') or ''),
        'widgets': payload_widgets, 'updated_at': time.strftime('%Y-%m-%d %H:%M:%S'),
```

```python
def share(self, request, token=None):
    """分享访问：按 share_url 中的 token 取已发布大屏。"""
    row = BiScreen.objects.filter(share_url__endswith=f'/{token}', status=1, is_deleted=0).first()
```

关键点：

1. **`share_password` 完全没校验**：模型有该字段、前端编辑弹窗可填，但 `share` 接口既不读取也不校验。
2. **接口权限仍是 `PermissionCheck`**：`CrudViewSet.permission_classes` 固定返回 `[PermissionCheck]`，而 `PermissionCheck` 继承 `LoginRequired`，匿名访问会抛 401。对比 3D 分享接口显式使用 `AllowAny`（`backend/apps/scene3d/views.py:820`），BI 这边没有做免登录覆盖。
3. **重复发布会换 token**：`publish` 每次都生成新 `uuid`，旧分享链接立即失效。
4. `share_url__endswith` 依赖完整路径后缀，若 `share_url` 被手工改成别的结构，匹配会失效。

## 模块小结

- 报表与 BI 共用一套「配置 JSON + 公共图表数据构建」的渲染思路：`rp_template.template_json` / `bi_screen.screen_config` 都是长文本 JSON，后端解析后统一产出 `chart/categories/series/columns/rows`。
- 所有列表接口都走 `CrudViewSet` + `StandardPagination`，返回 `{code,msg,data:{total,page,page_size,list}}`，与 Spring 的 `PageInfo` 结构对齐。
- 报表生成会落 `rp_generate_record`，大屏访问会落 `bi_visit_log`，属于“主表 + 日志表”的典型设计。
- 阅读时需要特别注意的实现差异：
  1. **BI 分享未真正免登录**：前端守卫与后端 `PermissionCheck` 都要求登录，和「免登录分享」的产品语义不一致；3D 分享已用 `AllowAny`，可作对照。
  2. **`share_password` 未生效**：字段可配置但分享接口不校验。
  3. **报表生成权限**：前端按钮要求 `rp:generate`，后端 `perm_map` 未覆盖 `generate`，实际只校验登录。
  4. **回放使用模板当前配置**：`rp_generate_record` 没有保存 JSON 快照，模板改动后历史回放结果会变化；逻辑删除的模板仍会被查到。
  5. **重复的 `'list'` extra**：`report/bi/outer` 的 `urls.py` 都在 `crud()` 基础上又追加了 `'list'`，因注册顺序而不生效。
  6. **数据集列表的 `screen_name`**：前端按 `screen_name || screen_id` 展示，但序列化器只返回 `screen_id`，名称永远不会出现。
