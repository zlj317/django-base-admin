```javascript
async function fetchList() {
```

3. 路由工厂 [urls.py#L7-L22](file:///d:/other/django-base-admin/backend/apps/scene3d/urls.py#L7-L22)：

```python
def crud(prefix, viewset, extras=None):
    """路由工厂：按统一约定生成 CRUD 与自定义动作。"""
    def _join(s):
        # 有前缀时拼成 "category/list"，空前缀时保持 "list"
        return f'{prefix}/{s}' if prefix else s
    # 一个 as_view 结果同时绑定 GET/POST/DELETE，按 HTTP 方法分派到 list/create/delete
    list_view = viewset.as_view({'get': 'list', 'post': 'create', 'delete': 'delete'})
    # 编辑只允许 PUT，避免误用 PATCH 造成部分更新语义
    item_view = viewset.as_view({'put': 'update'})
    patterns = [
        path(_join('list'), list_view),          # 分页列表
        path(_join('create'), list_view),        # 新增
        path(_join('delete'), list_view),        # 批量删除
        path(_join('update/<int:pk>'), item_view),  # 编辑指定记录
    ]
    # extras 是 { '路径后缀': {'HTTP方法': '动作名'} } 形式的字典
    for suffix, methods in (extras or {}).items():
        patterns.append(path(_join(suffix), viewset.as_view(methods)))
    return patterns
```

### 4. 视图内部业务逻辑

真正执行列表逻辑的是父类 `Scene3dCrudViewSet`（[views.py#L187-L278](file:///d:/other/django-base-admin/backend/apps/scene3d/views.py#L187-L278)）：

```python
class Scene3dCrudViewSet(viewsets.ViewSet):
    """3D 模块 CRUD 基类：在通用权限映射上追加自定义 action 的权限标识。"""

    model = None            # 子类指定模型
    serializer_class = None # 子类指定序列化器
    perm_prefix = 'scene3d' # 权限前缀
    filter_map = {}         # 查询参数映射
    order_by = '-id'        # 默认排序
    extra_perm_map = {}     # 自定义动作的权限标识

    @property
    def permission_classes(self):
        """按通用 CRUD + 自定义动作拼出 perm_map。"""
        prefix = self.perm_prefix
        # 每个子类实例按自己的前缀生成四个基础权限码
        self.perm_map = {
            'list': f'{prefix}:list',
            'create': f'{prefix}:create',
            'update': f'{prefix}:update',
            'delete': f'{prefix}:delete',
        }
        # 子类声明的自定义动作权限合并进来，例如 publish/share/confirm
        self.perm_map.update(self.extra_perm_map)
        # PermissionCheck 会在 dispatch 时读取 self.perm_map[action]
        return [PermissionCheck]

    def get_queryset(self):
        """默认过滤逻辑删除并按 order_by 排序。"""
        # 所有查询都强制带 is_deleted=0，避免子类某个动作忘了过滤
        return self.model.objects.filter(is_deleted=0).order_by(self.order_by)

    def get_serializer_class(self):
        # 固定返回子类声明的序列化器，不做动态切换
        return self.serializer_class

    def apply_filters(self, queryset, request):
        """按 filter_map 过滤。"""
        for param, lookup in self.filter_map.items():
            value = request.query_params.get(param)   # 从 URL 查询串取参数
            if value not in (None, ''):                # 空值忽略，避免 scene_name='' 把数据全过滤掉
                # 动态拼 lookup：scene_name__icontains / scene_status__exact
                queryset = queryset.filter(**{f'{param}__{lookup}': value})
        return queryset

    def list(self, request):
        """标准分页列表。"""
        # 第一步：过滤逻辑删除 + 用户条件
        queryset = self.apply_filters(self.get_queryset(), request)
        # 第二步：StandardPagination 先 count 再切片，最后用序列化器输出
        return StandardPagination(request).paginate(queryset, self.get_serializer_class())
```

`StandardPagination` 的返回结构固定为 `{total, page, page_size, list}`（[backend/common/pagination.py#L44-L58](file:///d:/other/django-base-admin/backend/common/pagination.py#L44-L58)）。

```python
class Scene3dSceneSerializer(serializers.ModelSerializer):
    # 列表页需要直接显示分类名/模板名，避免前端再发两次请求
    category_name = serializers.SerializerMethodField(read_only=True)
    template_name = serializers.SerializerMethodField(read_only=True)

    def get_category_name(self, obj):
        # 按 category_id 查分类表，过滤逻辑删除
        row = Scene3dCategory.objects.filter(id=obj.category_id, is_deleted=False).first()
        return row.category_name if row else None

    def get_template_name(self, obj):
        if not obj.template_id:   # 没选模板直接返回空
            return None
        row = Scene3dTemplate.objects.filter(id=obj.template_id, is_deleted=False).first()
        return row.template_name if row else None

    class Meta:
        model = Scene3dScene
        fields = '__all__'     # 输出全部物理字段 + 上面两个方法字段
```

注意：这是**逻辑外键 + N+1 查询**。当前实现每行会各查一次分类和模板，数据量大时应改为 `select_related` 或一次性批量查询。Java 里类似 JPA 的 `@ManyToOne(fetch = FetchType.EAGER)`，但这里没有真正的数据库外键约束。

```javascript
    return
  }

    ElMessage.warning('场景至少需要一个对象才能发布')
    return

    await sceneApi.publish(sceneId)
    ElMessage.success('发布成功')
    dirty.value = false       // 清除「有未保存修改」标记
    await loadScene()         // 重新拉详情，刷新状态与对象 ID
  } finally {
    publishing.value = false
```

### 4. 保存逻辑（逐行）

[views.py#L333-L362](file:///d:/other/django-base-admin/backend/apps/scene3d/views.py#L333-L362)：

```python
def save(self, request, pk=None):
    """保存场景草稿：更新配置 + 整体替换对象 + 生成版本快照。"""
    scene = self.get_queryset().filter(pk=pk).first()  # 只在未删除场景中查找
    if not scene:
        return fail('场景不存在', code=404)             # 业务 404，HTTP 仍是 200

    body = request.data or {}                          # DRF 已把 JSON 解析成 dict
    config = _json(body.get('scene_config'), None)     # 兼容前端把 JSON 传成字符串
    if config is not None and not isinstance(config, dict):
        return fail('scene_config 必须是对象', code=400) # 结构校验

    raw_objects = body.get('objects')                  # 原始对象数组入参
    objects = _json(raw_objects, None)                 # 解析成 Python list
    if raw_objects is not None and (
        objects is None or not isinstance(objects, list) or len(objects) > MAX_OBJECTS
    ):
        return fail(f'objects 必须是数组且不超过 {MAX_OBJECTS} 条', code=400)

    base = dict(_json(scene.scene_config, {}))         # 以数据库现有配置为底
    if config is not None:
        config.pop('share', None)                      # 分享令牌不允许前端覆盖
        base.update(config)                            # 浅合并：只覆盖提交的键
    size = len(json.dumps(base, ensure_ascii=False).encode('utf-8'))
    if size > MAX_CONFIG_SIZE:                         # 2MB 上限，防止 JSON 列过大
        return fail('场景配置超过 2MB，请精简后再保存', code=400)

    scene.scene_config = base                          # 赋值给模型字段
    if body.get('remark') is not None:
        scene.remark = str(body.get('remark'))[:1000]  # 描述可选更新
    scene.update_by = _username(request)               # 审计：更新人
    scene.update_time = _now()                         # 审计：更新时间
    scene.save()                                       # UPDATE scene3d_scene

    if objects is not None:
        _replace_objects(scene.id, objects, request)   # 整体替换场景对象

    _snapshot(scene, body.get('version_desc') or '手动保存场景配置', 0, request)
    return ok(_scene_payload(scene), '保存成功')       # 返回主表+对象+绑定+规则+授权
```

关键点：

1. `scene_config` 是**浅合并**，不是整体覆盖。这样前端只提交相机参数时，不会把数据库里的分享令牌冲掉。
2. `share` 字段被强制剔除，防止普通编辑用户伪造/覆盖分享令牌。
3. 对象采用「整体替换」而不是逐条增删，见 `_replace_objects`。

`_replace_objects`（[views.py#L150-L184](file:///d:/other/django-base-admin/backend/apps/scene3d/views.py#L150-L184)）：

```python
def _replace_objects(scene_id, rows, request):
    """用提交的 objects 数组整体替换场景对象：按 object_code 更新，缺失项逻辑删除。"""
    # 先把数据库现有对象按 object_code 建索引，便于 O(1) 判断新增还是更新
    existing = {row.object_code: row for row in
                Scene3dSceneObject.objects.filter(scene_id=scene_id, is_deleted=False)}
    keep_codes = set()  # 本次提交中仍然保留的 object_code

    for item in rows:
        code = str(item.get('object_code') or '').strip()[:50]
        if not code:
            continue  # 没有业务编码的对象直接跳过，无法做幂等

        resource_id = item.get('resource_id')
        # 资源必须真实存在，否则会产生指向已删除模型的脏对象
        if not resource_id or not Scene3dModelResource.objects.filter(
                id=resource_id, is_deleted=False).exists():
            continue

        keep_codes.add(code)
        row = existing.get(code)
        if not row:
            # 新对象：手工生成雪花主键，并记录创建审计
            row = Scene3dSceneObject(
                id=next_id(), scene_id=scene_id, object_code=code,
                create_by=_username(request), create_time=_now()
            )

        row.object_name = str(item.get('object_name') or code)[:100]
        row.resource_id = resource_id
        row.parent_id = item.get('parent_id') or None
        row.position = _vec(item.get('position'))   # 归一化成 {x,y,z}
        row.rotation = _vec(item.get('rotation'))
        row.scale = _vec(item.get('scale'), {'x': 1, 'y': 1, 'z': 1})
        row.label_text = (item.get('label_text') or '')[:200]
        row.is_show_label = 1 if item.get('is_show_label', 1) in (1, '1', True, 'true') else 0
        row.sort = int(item.get('sort') or 0)
        row.remark = (item.get('remark') or '')[:500]
        row.update_by = _username(request)
        row.update_time = _now()
        row.save()  # INSERT 或 UPDATE

    # 数据库里有、本次没提交的对象 => 用户删掉了，做逻辑删除
    stale = [row for code, row in existing.items() if code not in keep_codes]
    for row in stale:
        row.is_deleted = 1
        row.update_by = _username(request)
        row.update_time = _now()
        row.save(update_fields=['is_deleted', 'update_by', 'update_time'])
    return len(keep_codes)
```

这里没有使用 `transaction.atomic()`。如果保存主表成功、替换对象中途异常，会出现主表配置已变、对象只改了一半的情况。可以把它理解为 Java 里一组 `repository.save()` 调用但没有 `@Transactional`。文档只描述现状，不修改业务代码。

`_snapshot` 生成版本（[views.py#L126-L148](file:///d:/other/django-base-admin/backend/apps/scene3d/views.py#L126-L148)）：

```python
def _snapshot(scene, desc, is_publish=0, request=None):
    """把场景主配置与对象数组写入版本快照。"""
    objects = []
    # 逐条导出对象，只保留回滚必需字段，避免快照里塞入审计字段
    for row in Scene3dSceneObject.objects.filter(
            scene_id=scene.id, is_deleted=False).order_by('sort', 'id'):
        objects.append({
            'object_name': row.object_name,
            'object_code': row.object_code,
            'resource_id': row.resource_id,
            'parent_id': row.parent_id,
            'position': row.position,
            'rotation': row.rotation,
            'scale': row.scale,
            'label_text': row.label_text,
            'is_show_label': row.is_show_label,
            'sort': row.sort,
            'remark': row.remark,
        })

    payload = {'scene_config': _json(scene.scene_config, {}), 'objects': objects}
    version = Scene3dSceneVersion.objects.create(
        id=next_id(),                              # 雪花主键
        scene_id=scene.id,
        version_no=_next_version(scene.id),        # v1/v2/v3...
        version_desc=desc[:500],
        snapshot_config=payload,                   # 整个快照存 JSON 列
        is_publish=is_publish,                     # 0 普通保存 / 1 发布版本
        create_by=_username(request) if request else '',
        create_time=_now(),
    )
    return version
```

```python
def publish(self, request, pk=None):
    """发布场景：校验对象与资源，写发布版本。"""
    scene = self.get_queryset().filter(pk=pk).first()
    if not scene:
        return fail('场景不存在', code=404)

    # 前置校验 1：场景必须有对象，否则发布出去是空场景
    objects = list(Scene3dSceneObject.objects.filter(scene_id=scene.id, is_deleted=False))
    if not objects:
        return fail('场景暂无对象，无法发布', code=400)

    # 前置校验 2：所有对象引用的模型资源必须仍然存在
    resource_ids = {obj.resource_id for obj in objects}
    valid = Scene3dModelResource.objects.filter(
        id__in=resource_ids, is_deleted=False).count()
    if valid < len(resource_ids):
        return fail('存在引用了已删除模型的场景对象，请先修复', code=400)

    scene.scene_status = 1          # 1 = 已发布
    scene.update_by = _username(request)
    scene.update_time = _now()
    scene.save()                    # UPDATE scene3d_scene SET scene_status=1

    # 发布成功的同时写一条 is_publish=1 的版本，后续可回滚
    _snapshot(scene, '发布场景', 1, request)
    return ok(self.serializer_class(scene).data, '发布成功')
```

```python
def preview(self, request, pk=None):
    """返回模拟读数（真实接入需实现对应数据源适配器）。"""
    bind = self.get_queryset().filter(pk=pk).first()  # 未删除的绑定记录
    if not bind:
        return fail('绑定不存在', code=404)

    config = _json(bind.source_config, {})            # source_config 可能是字符串
    field = (config.get('field') if isinstance(config, dict) else '') or 'value'

    # 用主键尾号做「稳定伪随机」：同一条绑定每次预览数值相同，便于联调
    seed = int(str(bind.id)[-3:]) if str(bind.id).isdigit() else 0

    if field in ('temperature', 'temp'):
        value = round(25 + (seed % 90) / 10, 1)       # 25.0 ~ 33.9
        unit = '℃'
    elif field in ('load', 'usage', 'percent', 'humidity'):
        value = round(30 + (seed % 60), 1)            # 30.0 ~ 89.9
        unit = '%'
    else:
        value = float(seed % 100)                     # 0 ~ 99
        unit = ''

    return ok({
        'bind_id': bind.id,
        'bind_name': bind.bind_name,
        'object_id': bind.object_id,
        'value': value,
        'unit': unit,
        'time': _now(),
        'source': bind.data_source_type,
        'note': '模拟读数；真实值需接入对应数据源适配器',  # 明确告诉前端这是假数据
    }, '预览成功')
```

重点：这里**没有执行任何 SQL 查询业务表**。它只是根据 `source_config.field` 和主键尾号造一个稳定的数字。真实接入数据库/API/WebSocket 需要一个适配器层，当前仓库未实现。

```python
    """新增前校验表名白名单，避免绑定到任意表。"""
    config = _json((request.data or {}).get('source_config'), {})
    table = config.get('table') if isinstance(config, dict) else None
```

```python
def stats(self, request):
    """统计近 N 天告警数量、未确认数与级别分布。"""
    try:
        days = int(request.query_params.get('days') or 7)  # 默认 7 天
    except (TypeError, ValueError):
        return fail('days 必须是整数', code=400)

    days = min(max(days, 1), 90)  # 限制在 1~90 天，防止一次统计整年

    start = (datetime.now() - timedelta(days=days)).strftime('%Y-%m-%d %H:%M:%S')
    # 先按时间窗口过滤，后面的 count 都在这个 QuerySet 上继续 filter
    qs = Scene3dAlarmRecord.objects.filter(is_deleted=False, create_time__gte=start)

    scene_id = request.query_params.get('scene_id')
    if scene_id:
        qs = qs.filter(scene_id=scene_id)  # 可选按场景统计

    # 三级各查一次 count；当前实现是 3 次 SQL，数据量大时可改成 group by
    by_level = {level: qs.filter(alarm_level=level).count() for level in (1, 2, 3)}

    return ok({
        'days': days,
        'total': qs.count(),                          # 总数
        'unconfirmed': qs.filter(alarm_status=0).count(),  # 未确认
        'confirmed': qs.filter(alarm_status=1).count(),    # 已确认
        'closed': qs.filter(alarm_status=2).count(),       # 已闭环
        'by_level': [
            {'level': 1, 'name': '提示', 'value': by_level[1]},
            {'level': 2, 'name': '警告', 'value': by_level[2]},
            {'level': 3, 'name': '严重', 'value': by_level[3]},
        ],
    }, '查询成功')
```

```python
def confirm(self, request, pk=None):
    """确认告警：记录确认人/时间与处置备注。"""
    row = self.get_queryset().filter(pk=pk).first()  # 未删除记录
    if not row:
        return fail('告警记录不存在', code=404)

    if row.alarm_status != 0:
        return fail('该告警已确认或已闭环', code=400)  # 幂等拦截，防止重复确认

    user = _user(request)                 # JWT 中间件挂到 request.user_info
    row.alarm_status = 1                  # 0 未确认 -> 1 已确认
    row.confirm_user_id = user.id if user else None  # 记录确认人
    row.confirm_time = _now()             # 确认时间

    if (request.data or {}).get('remark'):  # 备注可选
        row.remark = str(request.data.get('remark'))[:1000]

    row.save()                            # UPDATE scene3d_alarm_record
    return ok(self.serializer_class(row).data, '确认成功')
```

注意这里用 `row.save()` 会更新所有字段，不是 `update_fields=['alarm_status', ...]`。并发下两个请求同时确认同一条也可能都成功，因为没有行锁或条件更新。Java 里可用：

```sql
UPDATE ... SET alarm_status = 1 WHERE id = ? AND alarm_status = 0;
-- 影响行数为 0 则说明已被别人确认
```

当前代码没有这么做，属于可改进点。

### 5. 免登录分享逻辑

```python
class Scene3dShareView(views.APIView):
    """分享页数据接口：免登录，凭 token 读取已发布场景。"""

    permission_classes = [AllowAny]   # 覆盖全局默认认证/登录要求
    authentication_classes = []       # 不解析登录态，避免过期 token 导致 401

    def get(self, request, token):
        """按 token 返回场景详情，并校验有效期。"""
        scene = None

        # 遍历所有已发布场景，在 scene_config['share'] 里找 token
        # 已发布场景数量通常不大，所以直接遍历；数据量大时应把 token 单独建索引
        for row in Scene3dScene.objects.filter(is_deleted=False, scene_status=1):
            config = _json(row.scene_config, {})  # 兼容历史字符串
            share = _json(config.get('share'), {}) if isinstance(config, dict) else {}
            share = share if isinstance(share, dict) else {}

            if share.get('token') == token:      # 命中令牌
                scene = row
                expire = share.get('expire')     # 过期时间字符串
                if expire:
                    try:
                        if datetime.now() > datetime.strptime(
                                expire, '%Y-%m-%d %H:%M:%S'):
                            return fail('分享链接已过期', code=410)  # 410 已失效
                    except ValueError:
                        pass  # 时间格式异常时按有效处理
                break

        if not scene:
            return fail('分享链接无效或场景已下线', code=404)

        # access_type='分享链接' 会让 _scene_payload 写访问日志时标记来源
        return ok(_scene_payload(scene, request, '分享链接'), '查询成功')
```

`_scene_payload` 在返回详情的同时写访问日志（[views.py#L92-L118](file:///d:/other/django-base-admin/backend/apps/scene3d/views.py#L92-L118)）：

```python
def _scene_payload(scene, request=None, access_type=None):
    """组装场景详情：主表 + 对象 + 数据绑定 + 告警规则 + 授权列表。"""
    # 四张子表按场景 ID 查询，过滤逻辑删除
    objects = Scene3dSceneObject.objects.filter(
        scene_id=scene.id, is_deleted=False).order_by('sort', 'id')
    binds = Scene3dDataBind.objects.filter(
        scene_id=scene.id, is_deleted=False).order_by('id')
    rules = Scene3dAlarmRule.objects.filter(
        is_deleted=False, bind_id__in=[b.id for b in binds] or [0])
    perms = Scene3dScenePermission.objects.filter(
        scene_id=scene.id, is_deleted=False)

    data = Scene3dSceneSerializer(scene).data
    data['scene_config'] = _json(data.get('scene_config'), {})  # 统一成对象
    data['objects'] = Scene3dSceneObjectSerializer(objects, many=True).data
    data['bindings'] = Scene3dDataBindSerializer(binds, many=True).data
    data['rules'] = Scene3dAlarmRuleSerializer(rules, many=True).data
    data['permissions'] = Scene3dScenePermissionSerializer(perms, many=True).data

    if request is not None:
        # 取真实 IP：优先 X-Forwarded-For 第一段
        ip = request.META.get('HTTP_X_FORWARDED_FOR', '').split(',')[0].strip() or \
             request.META.get('REMOTE_ADDR', '')
        user = _user(request)
        Scene3dAccessLog.objects.create(
            id=next_id(),
            scene_id=scene.id,
            user_id=user.id if user else None,   # 匿名分享访问为 None
            access_ip=ip[:50],
            access_ua=(request.META.get('HTTP_USER_AGENT') or '')[:1000],
            access_type=access_type or '系统登录',
            stay_seconds=0,                       # 前端未上报停留时长
            access_time=_now(),
        )
    return data
```

### 6. 分享链接生成逻辑

[views.py#L478-L499](file:///d:/other/django-base-admin/backend/apps/scene3d/views.py#L478-L499)：

```python
def share(self, request, pk=None):
    """生成分享链接：token 写入场景配置，有效期 1~30 天。"""
    scene = self.get_queryset().filter(pk=pk).first()
    if not scene:
        return fail('场景不存在', code=404)

    if scene.scene_status != 1:
        return fail('仅已发布场景支持分享', code=400)  # 草稿不能分享

    try:
        days = int((request.data or {}).get('expire_days') or 7)  # 默认 7 天
    except (TypeError, ValueError):
        return fail('expire_days 必须是整数', code=400)

    if days < 1 or days > 30:
        return fail('有效期需在 1~30 天之间', code=400)

    token = uuid.uuid4().hex  # 32 位十六进制随机串
    expire = (datetime.now() + timedelta(days=days)).strftime('%Y-%m-%d %H:%M:%S')

    config = dict(_json(scene.scene_config, {}))
    # 令牌直接写进 scene_config JSON，不单独建分享表
    config['share'] = {'token': token, 'expire': expire, 'create_by': _username(request)}
    scene.scene_config = config
    scene.update_by = _username(request)
    scene.update_time = _now()
    scene.save()

    return ok({
        'token': token,
        'expire': expire,
        'share_url': f'/scene3d/share/{token}',
    }, '分享链接已生成')
```

注意：每次生成都会覆盖旧 token，旧链接立即失效。这是「一个场景同时只有一个有效分享链接」的设计。

### 10. 当前实现的注意点

- 分享 token 存在 JSON 里，查 token 需要遍历所有已发布场景，数据量大时性能差；建议单独建 `share_token` 索引表或 Redis。
