## 0. 先理解统一 CRUD 基类

后端绝大多数模块不写重复的增删改查，而是在 `apps/*/views.py` 里声明模型、序列化器、权限前缀和筛选字段，继承 `CrudViewSet`：

```python
class CrudViewSet(viewsets.ViewSet):
    """通用 CRUD。子类配置 model/serializer_class/perm_prefix/filter_map/order_by/module_name。"""

    model = None  # 子类必须指定模型类
    serializer_class = None  # 子类必须指定序列化器
    perm_prefix = ''  # 权限前缀，如 system:user，用于拼出各 action 的权限标识
    filter_map = {}  # 查询参数到 ORM lookup 的映射，声明后即可支持筛选
    order_by = 'id'  # 默认按编号升序，保证分页结果稳定
    module_name = ''  # 操作日志模块名，子类可覆盖为中文名

    @property
    def permission_classes(self):
        """按 perm_prefix 动态生成 perm_map，并固定使用 PermissionCheck。"""
        prefix = self.perm_prefix  # 取子类配置的权限前缀
        self.perm_map = {
            'list': f'{prefix}:list',  # 列表查询权限
            'create': f'{prefix}:create',  # 新增权限
            'update': f'{prefix}:update',  # 编辑权限
            'delete': f'{prefix}:delete',  # 删除权限
        }
        return [PermissionCheck]  # 只挂自定义权限类，登录校验已在其内部完成

    def get_queryset(self):
        """返回过滤逻辑删除后的基础查询集，供各类操作复用。"""
        return self.model.objects.filter(is_deleted=0).order_by(self.order_by)  # 逻辑删除条件在此统一收敛
```

列表统一走 `StandardPagination`：

```python
class StandardPagination:
    """基于 request 查询参数的分页器，输出 {total, page, page_size, list}。"""

    def paginate(self, queryset, serializer_class=None, many=True):
        """统计总数并截取当前页，可选序列化后按统一结构返回。"""
        total = queryset.count()  # 先取总数，前端需要据此计算总页数
        rows = queryset[self.offset:self.offset + self.page_size]  # 只查当前页数据，避免全量加载
        data = {
            'total': total,  # 总记录数
            'page': self.page,  # 当前页码
            'page_size': self.page_size,  # 每页条数
            'list': serializer_class(rows, many=many).data if serializer_class else list(rows),  # 有序列化器则输出字典
        }
        return ok(data)  # 统一包装成 {code, msg, data}
```

路由也用工厂函数复用 [system/urls.py](file:///d:/other/django-base-admin/backend/apps/system/urls.py#L7-L19)：

```python
def crud(prefix, viewset, extras=None):
    """生成标准 CRUD 路由：/list /create /update/<pk> /delete"""
    list_view = viewset.as_view({'get': 'list', 'post': 'create', 'delete': 'delete'})  # 集合
    item_view = viewset.as_view({'put': 'update'})  # 单条更新视图
    patterns = [
        path(f'{prefix}/list', list_view),  # 分页列表
        path(f'{prefix}/create', list_view),  # 新增
        path(f'{prefix}/delete', list_view),  # 批量删除
        path(f'{prefix}/update/<int:pk>', item_view),  # 按主键更新
    ]
    for suffix, methods in (extras or {}).items():  # 遍历调用方声明的自定义动作
        patterns.append(path(f'{prefix}/{suffix}', viewset.as_view(methods)))  # 追加自定义动作
    return patterns
```

Java 类比：`CrudViewSet` 类似 `BaseController<T, D>`，`filter_map` 类似 `QueryWrapper` 的字段映射，`StandardPagination` 类似 MyBatis-Plus 的 `Page<T>`。

```python
class UserViewSet(CrudViewSet):
    """用户管理视图：按动作切换序列化器，并在增改后同步角色。"""
    model = SysUser  # 目标模型
    perm_prefix = 'system:user'  # 权限标识前缀

    filter_map = {'username': 'icontains', 'nickname': 'icontains',  # 模糊筛选字段
                  'phone': 'icontains', 'status': 'exact', 'dept_id': 'exact'}  # 精确筛选字段

    def get_serializer_class(self):

        if self.action == 'list':
            return UserListSerializer  # 列表带部门与角色

        return UserCreateSerializer  # 新增含密码处理
```

基类 `list` 先 `apply_filters` 再分页：

```python
def apply_filters(self, queryset, request):
    """按 filter_map 把非空查询参数拼成 ORM 过滤条件。"""
    for param, lookup in self.filter_map.items():
        value = request.query_params.get(param)  # 取前端传入的筛选值
        if value not in (None, ''):
            queryset = queryset.filter(**{f'{param}__{lookup}': value})  # 动态拼接 ORM lookup
    return queryset
```

最终 SQL 近似：`SELECT ... FROM sys_user WHERE is_deleted=0 AND username LIKE %s ... ORDER BY id LIMIT 10 OFFSET 0`。

列表序列化器 [UserListSerializer](file:///d:/other/django-base-admin/backend/apps/system/serializers.py#L83-L115) 用 `SerializerMethodField` 做两次补充查询：

```python
class UserListSerializer(serializers.ModelSerializer):
    """用户列表序列化器：补充部门名、角色与前端使用的 0/1 状态。"""
    dept_name = serializers.SerializerMethodField()  # 只读：部门名称
    roles = serializers.SerializerMethodField()  # 只读：已分配角色
    status = serializers.SerializerMethodField()  # 只读：把布尔状态转成 0/1

    def get_status(self, obj):
        """把布尔状态转成前端约定的 0/1。"""
        return 1 if obj.status else 0  # True 视为 1

    def get_dept_name(self, obj):
        """按 dept_id 查询部门名称，未归属部门时返回空。"""
        if obj.dept_id:
            return SysDept.objects.filter(id=obj.dept_id).values_list('dept_name', flat=True).first()
        return None

    def get_roles(self, obj):
        """联表查询用户已分配的角色（过滤已删除角色）。"""
        with connection.cursor() as cursor:
            cursor.execute(
                "SELECT r.id, r.role_name FROM sys_user_role ur "
                "JOIN sys_role r ON r.id = ur.role_id AND r.is_deleted = 0 "
                "WHERE ur.user_id = %s", [obj.id])
            return [{'id': r[0], 'role_name': r[1]} for r in cursor.fetchall()]
```

注意 N+1：每行用户都会查一次部门与角色。数据量小可接受；若列表很大，应改成批量联表或 `prefetch`。

## 示例 7：新增用户并同步用户-角色关联

### 前端触发

[user/index.vue](file:///d:/other/django-base-admin/frontend/src/views/system/user/index.vue#L565-L602) 的 `handleSave()`，入口 `openCreate()`（524 行）：

```js
// 保存：先整表校验，再按 form.id 分流新增/修改
async function handleSave() {
  await formRef.value.validate()  // 校验失败会 reject，不进入提交
  saving.value = true
  try {
    if (form.id) {
      // 编辑：刻意排除 password，改密走独立接口
```

### 后端链路

`POST /api/system/user/create` → `UserViewSet.create`（基类 [CrudViewSet.create](file:///d:/other/django-base-admin/backend/common/views.py#L63-L92)），但序列化器由 `get_serializer_class()` 选为 `UserCreateSerializer`。

```python
def create(self, request):
    """新增接口：唯一性预校验 → 反序列化校验 → 补审计字段 → 落库 → 子类钩子。"""
    # 逻辑删除记录仍占用数据库唯一索引，新增前先给出明确提示，避免直接落库时报 500
    for field in self.model._meta.fields:
        if not field.unique or field.primary_key or not hasattr(self.model, 'is_deleted'):
            continue  # 只校验业务唯一字段
        value = request.data.get(field.attname)
        if value in (None, ''):
            continue
        qs = self.model.objects.filter(**{field.attname: value})  # 含已删除记录
        if qs.filter(is_deleted=0).exists():
            return fail(f'{field.verbose_name}已存在', code=400)
        if qs.exists():  # 仅被已删除记录占用，数据库唯一键仍会冲突
            return fail(f'{field.verbose_name}已被已删除记录占用，请更换{field.verbose_name}', code=400)
    serializer = self.get_serializer_class()(data=request.data)  # 绑定请求体做字段校验
    serializer.is_valid(raise_exception=True)  # 失败交给全局异常处理器转 400
    now = time.strftime('%Y-%m-%d %H:%M:%S')
    user = getattr(request, 'user_info', None)
    audit = {}
    if hasattr(self.model, 'create_by'):
        audit = dict(create_by=user.username if user else '', create_time=now,
                     update_by=user.username if user else '', update_time=now)
    instance = serializer.save(**audit)  # 落库并生成雪花主键
    self.after_create(request, instance)  # 子类钩子：同步角色关联
    return ok(self.get_serializer_class()(instance).data, '新增成功')
```

`UserViewSet.after_create`（[user.py](file:///d:/other/django-base-admin/backend/apps/system/views/user.py#L42-L45)）写关联表：

```python
def after_create(self, request, instance):
    """新增成功后同步写入角色关联。"""
    role_ids = request.data.get('role_ids') or []  # 前端提交的角色ID
    save_user_roles(instance.id, role_ids)  # 覆盖保存角色关联

def save_user_roles(user_id, role_ids):
    """覆盖式保存用户角色：先清空旧关联再批量插入。"""
    with connection.cursor() as cursor:
        cursor.execute("DELETE FROM sys_user_role WHERE user_id = %s", [user_id])
        if role_ids:
            cursor.executemany(
                "INSERT INTO sys_user_role (user_id, role_id) VALUES (%s, %s)",
                [(user_id, rid) for rid in role_ids])
```

```python
class UserCreateSerializer(serializers.ModelSerializer):
    """新增用户序列化器：密码只写不读，并做账号唯一性校验。"""
    password = serializers.CharField(write_only=True, required=False, allow_blank=True)

    class Meta:
        model = SysUser
        fields = ['id', 'username', 'password', 'nickname', 'phone', 'email', 'avatar', 'dept_id', 'status']

    def validate_username(self, value):
        """校验用户名在未删除记录中唯一。"""
        if SysUser.objects.filter(username=value, is_deleted=0).exists():
            raise serializers.ValidationError('用户名已存在')
        return value

    def validate_password(self, value):
        """密码统一走 bcrypt 哈希；为空时给默认密码。"""
        return hash_password(value) if value else hash_password('123456')
```

```python
def retrieve(self, request, pk=None):
    """查询单个用户详情：列表字段 + 角色ID，供编辑弹窗回填。"""
    user = self.get_queryset().filter(pk=pk).first()  # 查未删除用户
    if not user:
        return fail('用户不存在', code=404)  # 业务码 404，HTTP 仍 200
    data = dict(UserListSerializer(user).data)  # 复用列表序列化字段
    with connection.cursor() as cursor:
        cursor.execute("SELECT role_id FROM sys_user_role WHERE user_id = %s", [pk])
        data['role_ids'] = [row[0] for row in cursor.fetchall()]  # 供前端多选回填
    return ok(data)
```

### 数据表

`sys_user` 查主体；`sys_user_role` 查角色 id 数组。前端编辑弹窗的“角色”多选绑定 `role_ids`。

### 后端链路

`PUT /api/system/user/update/{id}` → 基类 [CrudViewSet.update](file:///d:/other/django-base-admin/backend/common/views.py#L94-L110) → 序列化器为 `UserUpdateSerializer`。

```python
def update(self, request, pk=None):
    """编辑接口：部分更新，并记录更新人与更新时间。"""
    instance = self.get_queryset().filter(pk=pk).first()  # 只在未删除数据中查找
    if not instance:
        return self._not_found()  # 业务码 404
    serializer = self.serializer_class(instance, data=request.data, partial=True)  # partial 允许只传变更字段
    serializer.is_valid(raise_exception=True)
    user = getattr(request, 'user_info', None)
    audit = {}
    if hasattr(self.model, 'update_by'):
        audit = dict(update_by=user.username if user else '',
                     update_time=time.strftime('%Y-%m-%d %H:%M:%S'))
    serializer.save(**audit)  # 审计字段由后端强制覆盖，前端传了也不生效
    self.after_update(request, instance)  # 子类钩子：role_ids 显式提交时同步角色
    return ok(self.serializer_class(instance).data, '修改成功')
```

`UserViewSet.after_update`（[user.py](file:///d:/other/django-base-admin/backend/apps/system/views/user.py#L47-L51)）：

```python
def after_update(self, request, instance):
    """编辑后仅在显式提交 role_ids 时更新角色。"""
    role_ids = request.data.get('role_ids')  # 未提交则为 None
    if role_ids is not None:  # 区分"未传"与"清空"
        save_user_roles(instance.id, role_ids)
```

### 数据表

- `sys_user`：更新昵称、手机、邮箱、部门、状态。
- `sys_user_role`：仅当 `role_ids` 出现在请求体时覆盖更新。

### Java 对照

`partial=True` ≈ `@PatchMapping` 或 `updateById` 只更新非空字段；`after_update` ≈ `@Transactional` 的关联同步；`role_ids is not None` ≈ `containsKey` 语义。

```python
@action(detail=True, methods=['put'], url_path='reset_pwd')
def reset_pwd(self, request, pk=None):
    """重置指定用户密码（仅更新密码与审计字段）。"""
    user = self.get_queryset().filter(pk=pk).first()
    if not user:
        return fail('用户不存在', code=404)
    password = request.data.get('password')
    if not password:

    return ok(None, '密码重置成功')
```

`update_fields` 是 Django 的局部更新优化：只生成包含这三列的 UPDATE 语句，等价于 MyBatis 的 `updateById` 选择性更新。

```python
def delete(self, request):
    """批量删除接口：按 ids 做逻辑删除。"""
    ids = request.data.get('ids') or []
    if not ids:
        return ok(None, '请选择要删除的数据')  # 未选数据给出提示而非报错
    user = getattr(request, 'user_info', None)

    return ok(None, '删除成功')
```

```python
def build_tree(rows, id_key='id', parent_key='parent_id', children_key='children'):
    """把扁平行列表组装成树；父节点缺失或自引用时按根节点处理。"""
    node_map = {r[id_key]: dict(r, children=[]) for r in rows}  # 先建索引并初始化 children
    roots = []  # 根节点集合
    for row in rows:
        parent_id = row.get(parent_key) or 0  # 0/None 视为顶级
        node = node_map[row[id_key]]  # 取当前节点
        parent = node_map.get(parent_id)  # 找父节点
        if parent is None or parent_id == row[id_key]:  # 父不存在或自己指向自己
            roots.append(node)  # 作为根节点
        else:
            parent['children'].append(node)  # 挂到父节点下
    return roots
```

```python
class MenuSerializer(serializers.ModelSerializer):
```

```python
def delete(self, request, pk=None):
    """单删：存在子菜单时禁止删除。"""
    menu = self.get_queryset().filter(pk=pk).first()
    if not menu:
        return fail('菜单不存在', code=404)
    if self.get_queryset().filter(parent_id=pk).exists():
        return fail('存在子菜单，不允许删除')
    menu.is_deleted = 1
    menu.save(update_fields=['is_deleted'])
    return ok(None, '删除成功')
```

同一路径还注册了批量入口 `DELETE /api/system/menu/delete`（不带 pk），由基类 `delete` 处理；因此视图方法 `delete(self, request, pk=None)` 用可选参数同时兼容两种调用。

### 数据表

`sys_menu.is_deleted`。删除父节点前必须先把子节点删掉或移走，否则会破坏树结构。

### Java 对照

`pk=None` 兼容单删/批删 ≈ 同一个 Controller 方法用 `@PathVariable(required=false)`；子节点校验 ≈ Service 层 `countByParentId`。
