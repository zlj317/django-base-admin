```python
def crud(prefix, viewset, extras=None):
    def _join(s): return f'{prefix}/{s}' if prefix else s
    list_view = viewset.as_view({'get': 'list', 'post': 'create', 'delete': 'delete'})  # 三个动作共用
    item_view = viewset.as_view({'put': 'update'})                                      # 更新单独注册
    patterns = [
        path(_join('list'), list_view),                # /api/list
        path(_join('create'), list_view),              # /api/create，POST 触发 create
        path(_join('delete'), list_view),              # /api/delete，DELETE 触发 delete
        path(_join('update/<int:pk>'), item_view),     # /api/update/{id}，PUT 触发 update
    ]
    for suffix, methods in (extras or {}).items():     # test/call/status 等额外动作
        patterns.append(path(_join(suffix), viewset.as_view(methods)))
    return patterns

urlpatterns = crud('api', views.OuterApiConfigViewSet, {
    'test/<int:pk>': {'post': 'test'},       # 连通性测试
    'call/<int:pk>': {'post': 'call'},       # 正式调用
    'status/<int:pk>': {'put': 'status'},    # 启用/停用
})
```

```python
# backend/common/views.py:63-92
def create(self, request):
    """新增接口：校验入参后补审计字段并回写完整对象。"""
    # 逻辑删除记录仍占用数据库唯一索引，新增前先给出明确提示，避免直接落库时报 500
    for field in self.model._meta.fields:
        if not field.unique or field.primary_key or not hasattr(self.model, 'is_deleted'):
            continue                                   # 只校验业务唯一字段，跳过主键
        value = request.data.get(field.attname)        # 取提交的唯一字段值
        if value in (None, ''):
            continue                                   # 未提交则交给序列化器处理
        qs = self.model.objects.filter(**{field.attname: value})  # 查询包含已删除记录
        if qs.filter(is_deleted=0).exists():
            return fail(f'{field.verbose_name}已存在', code=400)   # 未删除记录已占用
        if qs.exists():
            return fail(f'{field.verbose_name}已被已删除记录占用，请更换{field.verbose_name}', code=400)
    serializer = self.get_serializer_class()(data=request.data)  # 绑定请求体，DRF 做字段级校验
    serializer.is_valid(raise_exception=True)                    # 失败抛 ValidationError → 全局 400
    now = time.strftime('%Y-%m-%d %H:%M:%S')                     # 统一时间格式
    user = getattr(request, 'user_info', None)                   # JWT 认证类挂上的当前用户
    audit = {}
    if hasattr(self.model, 'create_by'):
        audit = dict(
            create_by=user.username if user else '',             # 创建人账号
            create_time=now,                                     # 创建时间
            update_by=user.username if user else '',             # 初始更新人
            update_time=now,                                     # 初始更新时间
        )
    instance = serializer.save(**audit)                          # 落库，雪花主键由字段默认值生成
    self.after_create(request, instance)                         # 子类扩展钩子，本类未覆盖
    return ok(self.get_serializer_class()(instance).data, '新增成功')  # 返回完整对象
```

结合实际模型，需要知道：

1. `OuterApiConfig` 只有 `id` 是唯一的（`id` 是主键，循环里被 `field.primary_key` 跳过），所以唯一性预检查对本接口**什么都不做**；`api_name`、`api_url` 允许重复。
2. `OuterApiConfigSerializer` 是 `fields='__all__'`（`backend/apps/outer/serializers.py:7-12`），DRF 会校验 `method`/`auth_type` 的 `choices`、`max_length`，但**不会校验** `headers_json`、`params_json`、`auth_config` 是否是合法 JSON——前端规则也只要求必填名称和 URL。写错 JSON 时，调用阶段 `_do_request` 会静默当成 `{}`。
3. 状态字段 `status` 默认 0；前端默认提交 1（启用），因此列表会出现“启用”的开关状态。

```python
# backend/apps/outer/views.py:30-91
def _do_request(self, config_obj, extra_params=None):
    """内部 helper：执行一次 HTTP 请求并写日志，返回 (ok_response 或 fail_response, http_code)。"""
    import urllib.request                        # 标准库 HTTP 客户端，无需额外依赖
    import urllib.error
    start = time.time()                          # 开始计时
    method = (config_obj.method or 'GET').upper()# 方法统一大写，缺省 GET
    url = config_obj.api_url or ''               # 目标地址
    headers = {}
    if config_obj.headers_json:
        try:
            headers = json.loads(config_obj.headers_json)   # 自定义请求头
        except Exception:
            headers = {}                          # JSON 非法时降级为空，避免直接 500
    data = {}
    if config_obj.params_json:
        try:
            data = json.loads(config_obj.params_json)       # 默认参数
        except Exception:
            data = {}
    if extra_params:
        data.update(extra_params)                # call 接口可覆盖同名参数
    if method in ('GET', 'DELETE') and data:
        from urllib.parse import urlencode, urlsplit, urlunsplit
        parts = urlsplit(url)                    # 拆 scheme/netloc/path/query/fragment
        qs = urlencode(data)                     # 字典 → a=1&b=2
        sep = '&' if parts.query else '?'        # 原 URL 已有查询串则用 & 追加
        url = urlunsplit((parts.scheme, parts.netloc, parts.path,
                          parts.query + sep + qs, parts.fragment))
        body = None                              # GET/DELETE 不带请求体
    else:
        body = json.dumps(data).encode('utf-8') if data else None  # POST/PUT 用 JSON 体
    http_code = None
    resp_body = ''
    error_msg = ''
    try:
        req = urllib.request.Request(url, data=body, headers=headers, method=method)
        with urllib.request.urlopen(req, timeout=10) as resp:      # 固定 10 秒超时
            http_code = resp.status                                # HTTP 状态码
            resp_body = resp.read().decode('utf-8', errors='replace')  # 响应体，容错解码
    except urllib.error.HTTPError as e:
        http_code = e.code                                         # 4xx/5xx 仍记录状态码
        try:
            resp_body = e.read().decode('utf-8', errors='replace') # 错误响应体尽量保留
        except Exception:
            resp_body = ''
        error_msg = str(e.reason) if hasattr(e, 'reason') else str(e)
    except Exception as e:
        error_msg = str(e)                                         # 连接失败/超时等
    cost_ms = int((time.time() - start) * 1000)                    # 毫秒耗时
    now = time.strftime('%Y-%m-%d %H:%M:%S')
    OuterApiRequestLog.objects.create(                             # 无论成败都落日志
        api_id=config_obj.id, api_name=config_obj.api_name,
        request_url=url, request_param=json.dumps(data, ensure_ascii=False),
        response_body=resp_body, http_code=http_code,
        cost_time=cost_ms, error_msg=error_msg,
        request_time=now, create_time=now,
    )
    if http_code and 200 <= http_code < 300:                       # 2xx 视为成功
        return ok({'http_code': http_code, 'cost_ms': cost_ms,
                   'body': resp_body[:1000]}, '调用成功'), http_code
    return fail(f'调用失败: {error_msg or "HTTP " + str(http_code)}',
                http_code or 0), http_code                         # 失败响应
```

```python
def call(self, request, pk=None):

    extra = (request.data or {}).get('params') or {}   # 请求体 params 作为覆盖参数
    resp, _ = self._do_request(row, extra)             # 合并到默认 params
```

必须知道的实现限制：

1. **`auth_type` / `auth_config` 没有被使用**：不管配置的是 token 还是 basic，`_do_request` 都不会拼 `Authorization` 头。需要鉴权的接口只能把 token 手工写进 `headers_json`。
2. **没有设置 `Content-Type: application/json`**：`urllib` 对带 body 的请求不会自动加 JSON 头，若第三方严格要求 JSON，需要在 `headers_json` 里自行配置。
3. **超时固定 10 秒**，没有连接超时/读超时区分，也不能按接口配置。
4. **URL 里不能做路径变量替换**：`api_url` 只是原样请求，`extra_params` 全部进 query 或 body。
5. 日志写入本身没有 try/except：若日志表异常（例如表不存在），测试接口会 500。

```python
# backend/apps/outer/views.py:113-136
class OuterApiRequestLogViewSet(CrudViewSet):
    """外部API调用日志表 CRUD：已发生的接口调用记录。"""
    model = OuterApiRequestLog
    serializer_class = OuterApiRequestLogSerializer
    perm_prefix = 'outer'
    filter_map = {'api_id': 'exact', 'api_name': 'icontains', 'http_code': 'exact'}
    order_by = '-id'

    def get_queryset(self):
        return self.model.objects.order_by(self.order_by)  # 日志表无 is_deleted，不能套基类过滤

    def list(self, request):
        """列表查询：OuterApiRequestLog 不继承 BaseModel。"""
        queryset = OuterApiRequestLog.objects.all().order_by('-id')  # 全量日志
        for param, lookup in self.filter_map.items():
            value = request.query_params.get(param)
            if value not in (None, ''):
                queryset = queryset.filter(**{f'{param}__{lookup}': value})
        return StandardPagination(request).paginate(queryset, OuterApiRequestLogSerializer)
```

```python
# backend/apps/sys_no/views.py:24-47
@action(detail=True, methods=['post'])
def generate(self, request, pk=None):
    """模拟生成一个编号：拼接前缀+日期+流水号并记录日志。"""
    row = SysNoRule.objects.filter(pk=pk, is_deleted=0).first()  # 只查未删除规则
    if not row:
        return fail('规则不存在', code=404)
    import datetime
    now = datetime.datetime.now()
    row.seq_current = (row.seq_current or 0) + row.step          # 读取-加步长-写回，非原子操作
    row.save(update_fields=['seq_current', 'update_time'])       # 只更新两列
    date_part = now.strftime('%Y%m%d')                           # 日期段固定 YYYYMMDD
    seq = str(row.seq_current).zfill(row.seq_length or 4)        # 左侧补零到 seq_length
    sep = row.separator or ''                                    # 分隔符
    no_value = f"{row.prefix}{sep}{date_part}{sep}{seq}"         # 前缀+日期+流水
    user = getattr(request, 'user_info', None)
    SysNoLog.objects.create(                                     # 每次生成都写日志
        rule_id=row.id,
        no_value=no_value,
        biz_type=request.data.get('biz_type'),                   # 前端未传 → None
        biz_id=request.data.get('biz_id'),                       # 前端未传 → None
        gen_user_id=user.id if user else None,
        gen_time=now,
    )
    return ok({'no_value': no_value}, '生成成功')                  # 键名是 no_value
```

实现与注释/前端预期的差异：

1. **不是原子自增**：两条并发请求可能都读到同一个 `seq_current`，各自 +1 后写回，产生重复编号。要做原子性需要 `select_for_update()` 行锁、数据库序列或 Redis `INCR`。
2. **`date_format` 被忽略**：模型有 `date_format='YYYYMMDD'` 字段，代码固定用 `%Y%m%d`。
3. **`reset_type` / `reset_time` 被忽略**：日/月/年重置逻辑未实现，流水号会一直累加。
4. **`status` 未校验**：停用的规则也能生成编号。
5. **`update_time` 实际不会刷新**：`save(update_fields=['seq_current', 'update_time'])` 只把内存里的旧值写回；代码没有先执行 `row.update_time = now`（该字段也没有 `auto_now`）。
6. **返回键是 `no_value`**，前端读 `res.data.number` 取不到，会退化成 `${res.data}`，页面显示 `[object Object]`。

#### 视图业务逻辑

```python
# backend/apps/imprint/views.py:39-51
class PrintTemplateViewSet(CrudViewSet):
    """打印模板表 CRUD。"""
    model = PrintTemplate
    serializer_class = PrintTemplateSerializer
    perm_prefix = 'imprint'                 # imprint:list/create/update/delete
    filter_map = {
        'template_name': 'icontains',       # 模板名称模糊
        'template_key': 'icontains',        # 模板编码模糊
        'biz_type': 'exact',                # 业务类型精确
        'status': 'exact',                  # 状态精确
    }
    order_by = '-id'                        # 新建模板在前
    module_name = '打印模板管理'
```

本类没有覆写 `list`/`get_queryset`，因此完全走基类：

1. `get_queryset()`：`PrintTemplate.objects.filter(is_deleted=0).order_by('-id')`。
2. `apply_filters()`：按四个筛选项动态拼 `icontains` / `exact`。
3. `StandardPagination`：读 `page`、`page_size`，返回 `total/page/page_size/list`。
4. 权限：`perm_map` 由 `perm_prefix='imprint'` 生成，列表要求 `imprint:list`。

由于 `PrintTemplate` 继承 `BaseModel`，新增/编辑会自动补 `create_by/create_time/update_by/update_time`；`template_key` 有 `unique=True`，基类 `create` 的预检查会返回明确的“已存在”提示，而不是数据库 500。这是它与示例 80（列配置没有唯一字段）的一个明显区别。

## 模块小结

- 三个模块都建立在 `CrudViewSet` 之上：`urls.py` 用 `crud()` 批量注册 `list/create/delete/update/status`，视图只声明 `model/serializer/filter_map/order_by/perm_prefix`。可以把它理解为“一个泛型 Controller 基类 + 每个资源一份配置”。
- 外部接口模块的价值在于**把第三方调用收到服务端**：前端只发一个 `POST /outer/api/test/{id}`，由 `_do_request` 用 `urllib` 发起真实请求并写日志。但鉴权配置（`auth_type`/`auth_config`）目前是“存而不用”，真正能鉴权的只有 `headers_json`。
- 编码规则模块的生成逻辑是**读-改-写**，没有行锁/事务，且忽略 `date_format` 与 `reset_type`；高并发下会重复取号。返回字段名 `no_value` 与前端读取的 `number` 也不一致。
- 导入模板列使用自声明的 `is_deleted`，不继承 `BaseModel`，因此审计字段缺失、排序被 `order_by('-id')` 覆盖。
- 打印模板是三个资源里模型最“标准”的一个：继承 `BaseModel`、有唯一键，基类的查重与审计逻辑都能正常生效。

阅读时需要特别注意的实现差异：

1. `outer` 的 `/api/test` 响应键是 `http_code/cost_ms/body`，前端读 `cost_time/response_body`，耗时永远显示 `-`。
2. `sys_no` 的 `/generate` 返回 `{no_value}`，前端读 `number`，弹窗会显示 `[object Object]`。
3. `sys_no` 的并发取号没有原子保护，且停用规则仍可生成。
4. `imprint/column` 的 `status/<pk>` 路由指向模型上不存在的 `status` 列，调用不会真正改状态。
5. `outer/sync` 的 `run_once` 是占位实现，返回“未实际拉取”，不要据此判断同步能力已落地。
6. `outer/log` 的 `urls.py` 重复注册了 `'list'`（`report/gen`、`bi/dataset` 也有同类问题），因注册顺序先命中基础 `list_view` 而不生效，不影响功能但会增加维护困惑。
