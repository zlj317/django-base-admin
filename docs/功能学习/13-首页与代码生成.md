两个底层工具：

```python
def _raw_count(table, where=''):
    """用原生 SQL 查某表的 count，失败返回 None。"""
    from django.db import connection
    try:
        with connection.cursor() as cursor:
            sql = f'SELECT COUNT(*) FROM {table}'
            if where:
                sql += f' WHERE {where}'
            cursor.execute(sql)
            row = cursor.fetchone()
            return row[0] if row else 0
    except Exception:
        return None  # 表不存在/字段不存在都返回 None，由调用方兜底


def _raw_query(sql, params=None, limit=None):
    """执行原生 SQL 并返回 dict 列表，失败返回空列表。"""
    from django.db import connection
    try:
        if limit:
            sql = f'{sql} LIMIT {int(limit)}'
        with connection.cursor() as cursor:
            cursor.execute(sql, params or [])
            # cursor.description 是列名元数据，用它把 tuple 结果转成 dict
            columns = [col[0] for col in cursor.description] if cursor.description else []
            return [dict(zip(columns, row)) for row in cursor.fetchall()]
    except Exception:
        return []
```

这种写法的优点：首页不会因为某张表还没建、某个字段改名而整体 500。缺点：SQL 散落在视图里，字段名没有 ORM 校验，改表结构时不会报编译错误。

### 3. 核心逻辑：7 日趋势

`rich` 内部定义了一个闭包 `_trend`（[home/views.py#L244-L262](file:///d:/other/django-base-admin/backend/apps/home/views.py#L244-L262)）：

```python
now = datetime.datetime.now()
today = now.date()
# 最近 7 天（含今天），固定 7 个点
dates = [today - datetime.timedelta(days=6 - i) for i in range(7)]

def _day_key(value):
    """把数据库返回的日期/时间值统一裁剪成 YYYY-MM-DD，便于做字典键。"""
    return str(value or '')[:10]

def _trend(table, column, where=''):
    """统计最近 7 天某张表按 column 日期的记录数。"""
    start = dates[0].strftime('%Y-%m-%d')  # 窗口起点
    # table/column 来自本文件常量，不接受用户输入，无注入风险
    sql = f'SELECT DATE({column}) AS d, COUNT(*) AS c FROM {table} WHERE {column} >= %s'
    if where:
        sql += f' AND {where}'  # 如 is_deleted=0
    sql += f' GROUP BY DATE({column})'  # 按天聚合
    rows = _raw_query(sql, params=[start])
    # 数据库只返回“有数据的天”，这里先建桶，再按 dates 顺序补 0
    bucket = {_day_key(r.get('d')): int(r.get('c') or 0) for r in rows}
    return [bucket.get(d.strftime('%Y-%m-%d'), 0) for d in dates]
```

```python
def list_db_tables(keyword=None):
    """读取当前库中的业务表（排除 sys_ 前缀与 Django 内置表）。"""
    with connection.cursor() as cursor:
        cursor.execute(
            "SELECT table_name, table_comment, table_rows, create_time "
            "FROM information_schema.tables "
            "WHERE table_schema = DATABASE() AND table_type = 'BASE TABLE' "
            "ORDER BY table_name")
        tables = cursor.fetchall()

    # 已导入表名集合：前端据此把对应行标为「已导入」，避免重复导入
    imported = set(SysGenTable.objects.filter(is_deleted=0)
                   .values_list('table_name', flat=True))

    result = []
    for name, comment, rows, create_time in tables:
        if name.startswith(('sys_', 'django_', 'auth_')):  # 系统表跳过
            continue
        if keyword and keyword.lower() not in name.lower():  # 关键字包含匹配
            continue
        result.append({
            'table_name': name,
            'table_comment': comment or '',
            'table_rows': rows or 0,
            'create_time': create_time.strftime('%Y-%m-%d %H:%M:%S')
                           if isinstance(create_time, datetime) else create_time,
            'imported': name in imported,  # 前端据此禁用「已导入」
        })
    return result
```

```python
def import_tables(self, request):
    """批量导入选中表：逐张处理并汇总成功与失败信息。"""
    table_names = request.data.get('tables') or []
    if not table_names:
        return fail('请选择要导入的表')

    errors, success = [], []
    username = request.user_info.username  # JWT 中间件挂载
    for name in table_names:
        try:
            # engine.import_table 会读表结构、推断字段配置，
            # 写入 sys_gen_table + sys_gen_table_column
            engine.import_table(name, username)
            success.append(name)
        except Exception as exc:
            # 单表失败不影响其它表，返回“表名: 原因”
            errors.append(f'{name}: {exc}')

    msg = f'成功导入 {len(success)} 张表'
    if errors:
        msg += '；失败：' + '；'.join(errors)
    return ok({'success': success, 'errors': errors}, msg)
```

```python
def download(self, request, pk=None):
    """打包 ZIP 下载；若配置了父级菜单则自动注册菜单与按钮权限。"""
    table, columns = self._load(pk)  # 读主表 + 字段配置
    if not table:
        return fail('配置不存在', code=404)

    # 只有配置了父级菜单才自动注册 sys_menu；避免下载动作产生副作用
    if table.parent_menu_id:
        self.create_menus(table, columns, request.user_info.username)

    # 渲染全部文件（9 个：models/serializers/views/urls/api js/vue...）
    files = engine.render_files(table, columns)

    buf = io.BytesIO()  # 内存 ZIP，不落临时文件
    with zipfile.ZipFile(buf, 'w', zipfile.ZIP_DEFLATED) as zf:
        for path, content in files:
            zf.writestr(path, content or '\n')  # 空内容补换行

    response = HttpResponse(buf.getvalue(),
                            content_type='application/octet-stream')
    # 文件名用模块英文名，如 student_code.zip
    response['Content-Disposition'] = \
        f'attachment; filename={table.module_name}_code.zip'
    return response
```

注意：这是**直接返回文件流**，不是 `ok()` 包装的 JSON。前端必须用 `responseType: 'blob'`，否则 axios 会把二进制当文本解析，得到乱码。

`create_menus`（[gen/views.py#L203-L237](file:///d:/other/django-base-admin/backend/apps/gen/views.py#L203-L237)）会：

- 用 `next_id()` 生成雪花主键；
- 原生 SQL 插入 `sys_menu`；
- 先插目录菜单（`menu_type='C'`），再插按钮（`menu_type='F'`）；
- 用 `perms = f'{module_name}:list'` 做幂等判断，重复下载不会重复注册。

### 7. 数据表与 ORM

| 表 | 作用 |
| --- | --- |
| `information_schema.tables` | 读库内所有基表 |
| `information_schema.columns` | 读列元数据 |
| `sys_gen_table` | 一条记录 = 一张待生成的业务表配置（`model` 见 [models.py#L14-L36](file:///d:/other/django-base-admin/backend/apps/gen/models.py#L14-L36)） |
| `sys_gen_table_column` | 一行 = 一个字段的表单/查询/导入导出配置（[models.py#L40-L77](file:///d:/other/django-base-admin/backend/apps/gen/models.py#L40-L77)） |
| `sys_menu` | 下载时自动注册菜单与按钮权限 |

`sys_gen_table` 关键字段：

- `table_name`：源表名，`unique=True` 防重复导入。
- `class_name`：生成实体类名，如 `BizStudent`。
- `module_name`：模块英文名，决定目录、权限前缀、ZIP 文件名。
- `module_comment`：模块中文名，用于菜单名。
- `url_prefix` / `gen_path`：生成的路由前缀与 Vue 页面路径。
- `parent_menu_id`：下载时菜单挂在哪个父节点。
- `is_delete_logic` / `import_enable` / `export_enable`：控制生成器是否产出逻辑删除、导入、导出代码。

`sys_gen_table_column` 关键字段：

- `gen_table_id`：逻辑外键 → `sys_gen_table.id`。
- `column_name` / `column_comment` / `column_type`。
- `is_pk`：主键列不参与表单/查询。
- `form_type`：input/number/date/select/radio/textarea/image/file。
- `is_list` / `is_query` / `query_type`：控制列表列与过滤方式。
- `is_required` / `is_export` / `is_import`。
- `sort`：字段顺序。

```javascript
// 导入：FormData 已由 api 层组装
async function submitImport() {
  if (!importFile.value) {
    ElMessage.warning('请选择要导入的文件')
    return
  }
  importing.value = true
  importErrors.value = []
  try {
    const result = await importApi(importFile.value)
    ElMessage.success(`成功导入 ${result.success_count} 条`)
    if (result.errors && result.errors.length) {
      importErrors.value = result.errors  // 保留弹窗展示错误行
    } else {
      importVisible.value = false
    }
    getList()
  } finally {
    importing.value = false
  }
}
```

```python
def import_data(self, request):
    """Excel 导入：校验失败返回行号与错误信息，不导入脏数据。"""
    file = request.FILES.get('file')  # 对应前端 FormData 的 'file'
    if not file:
        return fail('请上传 Excel 文件')

    try:
        wb = openpyxl.load_workbook(file, read_only=True)  # 只读，省内存
    except Exception:
        return fail('Excel 文件解析失败，请使用导入模板填写数据')

    ws = wb.active
    errors = []       # 全局错误列表
    to_create = []    # 待批量写入的模型实例
    for index, row in enumerate(ws.iter_rows(min_row=2, values_only=True), start=2):
        # 跳过表头；index 从 2 开始，正好是 Excel 行号
        if all(v in (None, '') for v in row):
            continue  # 空行跳过

        data = {}
        for pos, (name, _) in enumerate(IMPORT_COLUMNS):
            if pos < len(row):          # 兼容用户删列
                data[name] = row[pos]   # 按位置映射到模型字段

        row_errors = []
        if data.get('name') in (None, ''):
            row_errors.append('学生姓名 不能为空')  # 唯一必填校验

        if row_errors:
            errors.extend('第 %d 行：%s' % (index, e) for e in row_errors)
            continue  # 有错的行先跳过，全部行看完再决定整批是否导入

        to_create.append(BizStudent(**data))  # 先构造，不立即写库

    if errors:
        # 任何一行有错 => 整批拒绝，返回最多 50 条错误
        return ok({'success_count': 0, 'errors': errors[:50]},
                  '校验失败，未导入任何数据')

    user = getattr(request, 'user_info', None)
    now = time.strftime('%Y-%m-%d %H:%M:%S')
    for obj in to_create:
        obj.create_by = user.username if user else ''
        obj.create_time = now
    BizStudent.objects.bulk_create(to_create)  # 一次 INSERT 批量写入
    return ok({'success_count': len(to_create), 'errors': []}, '导入成功')
```

几个关键设计：

1. **整批拒绝**：只要有一行错误，一条都不写。避免用户看到“导入成功 90 条”却不知道哪 10 条失败。
2. **普通字段几乎不校验**：只校验 `name` 必填；`age` 传 `"abc"` 会在 `bulk_create` 时抛数据库异常（没有逐行 try），这一点与“整批拒绝”的初衷不完全一致。
3. **`bulk_create` 不走 `save()`**：不会触发模型的 `save` 钩子，也不会逐条生成 SQL，性能更好。
4. **没有事务显式包裹**：`bulk_create` 本身是单条 INSERT 多值，通常原子；但审计字段在 Python 循环里赋值，不会失败。

```python
def export_data(self, request):
    """按当前查询条件导出 Excel（限制最大行数）。"""
    # 复用列表过滤：前端把 query 原样传过来，导出结果与列表一致
    queryset = self.apply_filters(request, self.get_queryset())

    # 行数上限从系统参数读取，默认 50000
    queryset = queryset[:self.get_config_int('sys.export.max_rows', 50000)]

    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = '学生信息表'
    ws.append([label for _, label in EXPORT_COLUMNS])  # 中文表头

    for row in queryset:
        values = []
        for name, _ in EXPORT_COLUMNS:
            value = getattr(row, name, None)      # 按模型字段取值
            if isinstance(value, datetime):       # datetime -> 字符串
                value = value.strftime('%Y-%m-%d %H:%M:%S')
            values.append(value)
        ws.append(values)                         # 写一行

    buf = io.BytesIO()
    wb.save(buf)
    response = HttpResponse(
        buf.getvalue(),
        content_type='application/vnd.openxmlformats-officedocument.spreadsheetml.sheet')
    response['Content-Disposition'] = 'attachment; filename=' + FILE_NAME
    return response
```

```python
@staticmethod
def get_config_int(key, default):
    """读取系统配置中的整数项，缺失或非法时返回默认值。"""
    from apps.system.models import SysConfig  # 局部导入，避免循环依赖
    value = SysConfig.objects.filter(config_key=key, is_deleted=0) \
        .values_list('config_value', flat=True).first()
    try:
        return int(value) if value else default
    except (TypeError, ValueError):
        return default
```

### 8. 响应与前端消费

- 模板、导出都是文件流（`application/vnd...sheet` + `Content-Disposition: attachment`），前端用 Blob + `URL.createObjectURL` 下载。它们**不会**经过统一 `{code,msg,data}` 封装。
- 导入返回 JSON：

```json
{
  "code": 200,
  "msg": "导入成功",
  "data": { "success_count": 12, "errors": [] }
}
```

校验失败时：

```json
{
  "code": 200,
  "msg": "校验失败，未导入任何数据",
  "data": {
    "success_count": 0,
    "errors": ["第 3 行：学生姓名 不能为空", "第 8 行：学生姓名 不能为空"]
  }
}
```

前端 `submitImport` 根据 `result.errors.length` 决定保留弹窗还是关闭弹窗。

### 10. 当前实现的注意点

- 导入只校验姓名，其它字段类型错误会在 `bulk_create` 时整体抛异常，前端拦截器会弹通用错误，看不到具体行号。
- 导出用 `getattr(row, name)` 逐条读取，N 条记录会产生 N 次属性访问（ORM 已一次性取回，不是 N 次 SQL）。
- 模板文件名后端写死 `import_template.xlsx`，前端下载时改名；如果直接用浏览器打开 URL，会显示英文名。
- 导出行数上限读 `sys.export.max_rows`，配置不存在时默认 50000。
